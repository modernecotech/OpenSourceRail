from pathlib import Path
from hashlib import sha256
import sqlite3,json,subprocess,shutil,time
workspace=Path.home()/'.local/state/opensourcerail/continuous-design'
baseline='aa1070e8547a5bb079a602688cdabd65b5c318f013aced20f1a485d5a9b8ae6f'
epoch=workspace/'epochs'/baseline
assert json.loads((workspace/'active.json').read_text())['baseline_sha256']!=baseline
start=shutil.disk_usage(workspace).free
connection=sqlite3.connect((workspace/'queue.sqlite').as_uri()+'?mode=ro',uri=True)
rows=connection.execute("SELECT key,folder,receipt_sha FROM jobs WHERE baseline=? AND status='completed'",(baseline,)).fetchall();connection.close()
records=[];total=0;limit=4*1024**3
for key,folder,receipt_sha in rows:
 root=Path(folder).resolve()
 assert root.is_relative_to(epoch.resolve())
 receipt_path=root/'receipt.json';receipt_bytes=receipt_path.read_bytes()
 assert sha256(receipt_bytes).hexdigest()==receipt_sha
 receipt=json.loads(receipt_bytes)
 for name,digest in receipt['files_sha256'].items():
  p=root/name
  if p.is_symlink() or not p.is_file() or not p.resolve().is_relative_to(root) or p.suffix not in {'.json','.step'}:continue
  stat=p.stat()
  if stat.st_nlink!=1 or not 64*1024<=stat.st_size<=32*1024**2:continue
  if total+stat.st_size>limit:continue
  before_flags=subprocess.check_output(['/usr/sbin/filefrag','-v',str(p)],text=True)
  if 'encoded' in before_flags:continue
  usage=subprocess.check_output(['/usr/sbin/btrfs','filesystem','du','-s','--raw',str(p)],text=True).splitlines()[-1].split()
  if len(usage)<4 or int(usage[0])!=int(usage[1]) or int(usage[2])!=0:continue
  original=p.read_bytes();assert sha256(original).hexdigest()==digest
  backup=Path('/tmp/osr-compression-original-'+digest);backup.write_bytes(original)
  call=subprocess.run(['/usr/sbin/btrfs','filesystem','defragment','-czstd','-f',str(p)],capture_output=True,text=True,timeout=30)
  assert p.read_bytes()==original,'Original backup retained at '+str(backup)
  assert sha256(receipt_path.read_bytes()).hexdigest()==receipt_sha
  after_flags=subprocess.check_output(['/usr/sbin/filefrag','-v',str(p)],text=True)
  record=dict(job=key,path=p.relative_to(workspace).as_posix(),bytes=stat.st_size,sha256=digest,receipt_sha256=receipt_sha,
   exclusively_owned_extent_bytes=int(usage[1]),returncode=call.returncode,bytes_unchanged=True,encoded_extents_after='encoded' in after_flags,stderr=call.stderr)
  records.append(record);backup.unlink();total+=stat.st_size
  if len(records)%100==0:print('Verified compressed files',len(records),'logical bytes',total,flush=True)
  if call.returncode:break
 if total>=limit-32*1024**2:break
subprocess.run(['sync','-f',str(epoch)],check=True,timeout=30)
proof=dict(schema='osr-immutable-evidence-transparent-compression/1',baseline=baseline,observed_unix=time.time(),files=records,
 logical_bytes_processed=total,free_bytes_before=start,free_bytes_after=shutil.disk_usage(workspace).free,
 all_evidence_bytes_and_receipts_unchanged=all(r['bytes_unchanged'] for r in records),
 scope='Native Btrfs zstd compression of previously unencoded, exclusively owned historical receipt-bound text files only. No file deletion, filename/format/source/release change or compression of shared extents; filesystem free-space delta includes concurrent activity.')
Path('/tmp/osr-joint-evidence-compression.json').write_text(json.dumps(proof,indent=2)+'\n')
print(json.dumps(dict(files=len(records),logical_bytes_processed=total,all_bytes_unchanged=proof['all_evidence_bytes_and_receipts_unchanged'],observed_free_delta=proof['free_bytes_after']-start,free_gib=proof['free_bytes_after']/1024**3)),flush=True)
